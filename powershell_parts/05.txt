nt right,
        out int bottom
    )
    {
        left = 0;
        top = 0;
        right = 0;
        bottom = 0;

        IntPtr processHandle =
            IntPtr.Zero;

        IntPtr remoteBuffer =
            IntPtr.Zero;

        try
        {
            uint rights =
                PROCESS_VM_OPERATION |
                PROCESS_VM_READ |
                PROCESS_VM_WRITE |
                PROCESS_QUERY_INFORMATION;


            processHandle =
                OpenProcess(
                    rights,
                    false,
                    targetPid
                );


            if (
                processHandle ==
                IntPtr.Zero
            )
            {
                return false;
            }


            remoteBuffer =
                VirtualAllocEx(
                    processHandle,
                    IntPtr.Zero,
                    (UIntPtr)16,
                    MEM_COMMIT | MEM_RESERVE,
                    PAGE_READWRITE
                );


            if (
                remoteBuffer ==
                IntPtr.Zero
            )
            {
                return false;
            }


            IntPtr result =
                SendMessage(
                    toolbar,
                    TB_GETITEMRECT,
                    (IntPtr)index,
                    remoteBuffer
                );


            if (
                result ==
                IntPtr.Zero
            )
            {
                return false;
            }


            byte[] data =
                new byte[16];


            IntPtr bytesRead =
                IntPtr.Zero;


            bool readOK =
                ReadProcessMemory(
                    processHandle,
                    remoteBuffer,
                    data,
                    data.Length,
                    out bytesRead
                );


            if (!readOK)
            {
                return false;
            }


            if (
                bytesRead.ToInt64() <
                16
            )
            {
                return false;
            }


            left =
                BitConverter.ToInt32(
                    data,
                    0
                );


            top =
                BitConverter.ToInt32(
                    data,
                    4
                );


            right =
                BitConverter.ToInt32(
                    data,
                    8
                );


            bottom =
                BitConverter.ToInt32(
                    data,
                    12
                );


            if (
                right <= left ||
                bottom <= top
            )
            {
                return false;
            }


            return true;
        }

        finally
        {
            if (
                remoteBuffer !=
                IntPtr.Zero &&
                processHandle !=
                IntPtr.Zero
            )
            {
                VirtualFreeEx(
                    processHandle,
                    remoteBuffer,
                    UIntPtr.Zero,
                    MEM_RELEASE
                );
            }


            if (
                processHandle !=
                IntPtr.Zero
            )
            {
                CloseHandle(
                    processHandle
                );
            }
        }
    }

    public static bool ItemCenterScreen(
        IntPtr toolbar,
        int targetPid,
        int index,
        out int x,
        out int y
    )
    {
        x = 0;
        y = 0;

        int left;
        int top;
        int right;
        int bottom;


        bool ok =
            GetRemoteItemRect(
                toolbar,
                targetPid,
                index,
                out left,
                out top,
                out right,
                out bottom
            );


        if (!ok)
        {
            return false;
        }


        POINT p =
            new POINT();


        p.X =
            left +
            ((right - left) / 2);


        p.Y =
            top +
            ((bottom - top) / 2);


        if (
            !ClientToScreen(
                toolbar,
                ref p
            )
        )
        {
            return false;
        }


        x = p.X;
        y = p.Y;

        return true;
    }
