                    $title -match "Save"

                        $h

                    break
                }
            }

                $saveH -ne

            }

                $saveH
            )


        $saveTitle =
            [TFDStep13]::WindowText(

            )

        Write-Host (

        # STEP 13 END
        # ========================================================

        Write-Host ""

        Write-Host `

            -ForegroundColor Red


        Write-Host `

            "============================================" `
            -ForegroundColor Red

        Write-Host `
            $_.InvocationInfo.PositionMessage `
            -ForegroundColor Yellow

        throw   # 一鍵版：讓整個流程停下來
    }
