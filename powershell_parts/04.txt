
            )
        ) {

            (
                [System.Windows.Automation.InvokePattern]$invokePattern
            ).Invoke()

            return
        }


        # ========================================================
        # Delphi TevButton 實測沒有 InvokePattern
        #
        # 因此：
        # 先透過 UIA 精確找到「這顆控制項」
        # 再使用這顆控制項自己的即時 BoundingRectangle。
        #
        # 不是固定座標。
        # 不是比例座標。
        # ========================================================

        $rect =
            $Element.Current.BoundingRectangle


        if (
            $rect.Width -le 0 -or
            $rect.Height -le 0
        ) {

            throw (
                "控制項沒有 InvokePattern，且 BoundingRectangle 無效：" +
                $Element.Current.Name
            )
        }


        $x =
            [int][Math]::Round(
                $rect.Left +
                ($rect.Width / 2)
            )


        $y =
            [int][Math]::Round(
                $rect.Top +
                ($rect.Height / 2)
            )


        [TFDStep11Preview]::ClickPoint(
            $x,
            $y
        )
    }


    # ============================================================
    # MAIN
    # ============================================================

    try {

        Write-Host ""
        Write-Host "============================================" -ForegroundColor Cyan
        Write-Host "STEP 11：開啟場所紀錄表預覽" -ForegroundColor Cyan
        Write-Host "============================================" -ForegroundColor Cyan


        # ========================================================
        # 1. TFDSafeCln
        # ========================================================

        $proc =
            Get-Process TFDSafeCln |
            Select-Object -First 1


        if (
            $null -eq
            $proc
        ) {
            throw "找不到 TFDSafeCln.exe"
        }


        $pidNow =
            $proc.Id


        Write-Host "PID=$pidNow"


        # ========================================================
        # 2. 找 STEP 10 留下的場所紀錄表設定視窗
        # ========================================================

        $dialogH =
            Wait-Class `
                -ProcessId $pidNow `
                -ClassName "TSBMCheckPrintDialog" `
                -TimeoutSec 10


        Write-Host (
            "1. ✓ 找到場所紀錄表設定視窗"
        ) -ForegroundColor Green


        # ========================================================
        # 3. 重新 FromHandle
        # ========================================================

        $dialog =
            [System.Windows.Automation.AutomationElement]::FromHandle(
                $dialogH
            )


        if (
            $null -eq
            $dialog
        ) {
            throw "無法建立 TSBMCheckPrintDialog 的 UIA Element"
        }


        # ========================================================
        # 4. 找「確 認 」
        #
        # 實測：
        # Name='確 認 '
        # Class='TevButton'
        #
        # 因為 Name 內有空格，所以使用 Normalize 比對。
        # ========================================================

        $confirm =
            Find-NormalizedElement `
                -Root $dialog `
                -Name "確認" `
                -ClassName "TevButton"


        if (
            $null -eq
            $confirm
        ) {
            throw "找不到 TevButton『確 認 』"
        }


        Write-Host (
            "2. ✓ 找到確認：Name='{0}' Class='{1}'" -f
            $confirm.Current.Name,
            $confirm.Current.ClassName
        ) -ForegroundColor Green


        # ========================================================
        # 5. 前景化設定視窗
        # ========================================================

        [TFDStep11Preview]::SetForegroundWindow(
            $dialogH
        ) | Out-Null


        Start-Sleep -Milliseconds 200


        # ========================================================
        # 6. 按「確認」
        # ========================================================

        Invoke-IdentifiedElement `
            -Element $confirm


        Write-Host (
            "3. ✓ 已按『確認』"
        ) -ForegroundColor Green


        # ========================================================
        # 7. 等待 FastReport Preview
        # ========================================================

        $previewH =
            Wait-Class `
                -ProcessId $pidNow `
                -ClassName "TfrxPreviewForm" `
                -TimeoutSec 30 `
                -VisibleOnly


        if (
            $previewH -eq
            [IntPtr]::Zero
        ) {
            throw "TfrxPreviewForm 未出現"
        }


        Write-Host (
            "4. ✓ TfrxPreviewForm 已出現"
        ) -ForegroundColor Green


        # ========================================================
        # 8. 再驗證一次 Preview 仍存在
        # ========================================================

        Start-Sleep -Milliseconds 300


        $previewCheck =
            [TFDStep11Preview]::FindVisibleByClassPid(
                "TfrxPreviewForm",
                $pidNow
            )


        if (
            $previewCheck -eq
            [IntPtr]::Zero
        ) {
            throw "TfrxPreviewForm 曾出現，但驗證時已消失"
        }


        # ========================================================
        # STEP 11 END
        # ========================================================

        Write-Host ""
        Write-Host "============================================" -ForegroundColor Green
        Write-Host "★ STEP 11 成功" -ForegroundColor Green
        Write-Host "場所紀錄表預覽已開啟。" -ForegroundColor Green
        Write-Host "TfrxPreviewForm 已確認存在。" -ForegroundColor Green
        Write-Host "本步尚未操作 PDF 匯出。" -ForegroundColor Green
        Write-Host "============================================" -ForegroundColor Green
    }


    catch {

        Write-Host ""
        Write-Host "============================================" -ForegroundColor Red
        Write-Host "STEP 11 中止" -ForegroundColor Red
        Write-Host $_.Exception.Message -ForegroundColor Red
        Write-Host "============================================" -ForegroundColor Red

        Write-Host ""
        Write-Host $_.InvocationInfo.PositionMessage -ForegroundColor Yellow

        throw   # 一鍵版：讓整個流程停下來
    }
}
