# ============================================================
# 安管自動化 一鍵版
#
# 流程：場所建檔 → 查詢 → 人工選擇場所（唯一人工步驟）
#       → 場所紀錄表 PDF → 檢查紀錄表 PDF → 回主畫面
#       → 解析兩份 PDF 素材 → 固定格式 Word → 完成
#
# 使用方式（在 PowerShell 視窗）：
#   .\安管自動化_一鍵版.ps1
#   .\安管自動化_一鍵版.ps1 -Keyword 大巨蛋
#   .\安管自動化_一鍵版.ps1 -Keyword 大巨蛋 -RecentCheck 5 -RecentReport 2
#
# 沒給 -Keyword 會在開頭詢問。
# 任何一步失敗，整個流程立刻停止，不會繼續往下按。
# 各 STEP 內容沿用原本逐步測試成功的版本；
# 原 STEP 17（結構探測，只讀不操作）不在一鍵流程內。
# ============================================================

param(
    [string]$Keyword,
    [int]$RecentCheck = 5,
    [int]$RecentReport = 2
)

if ([string]::IsNullOrWhiteSpace($Keyword)) {
    $Keyword = Read-Host "請輸入場所名稱關鍵字"
}

if ([string]::IsNullOrWhiteSpace($Keyword)) {
    Write-Host "沒有輸入場所關鍵字，結束。" -ForegroundColor Red
    exit 1
}

if (-not (Get-Process TFDSafeCln -ErrorAction SilentlyContinue)) {
    Write-Host "找不到 TFDSafeCln（安管系統沒有開啟），結束。" -ForegroundColor Red
    exit 1
}

# 各 STEP 之間共用的資料
$TFD = @{
    Keyword      = $Keyword.Trim()
    RecentCheck  = [string]$RecentCheck
    RecentReport = [string]$RecentReport
    Choice       = $null
    PlaceNo      = $null
    PlaceName    = $null
    PlacePdf     = $null
    CheckPdf       = $null
    PlacePdfText   = $null
    CheckPdfText   = $null
    PlacePdfMethod = $null
    CheckPdfMethod = $null
    Parsed         = $null
    FinalDocx      = $null
    RetryChoice    = $false
}

$Steps = [ordered]@{}

# ============================================================
# STEP 1：進入場所建檔
# ============================================================
$Steps['1'] = {
    Add-Type -AssemblyName System.Windows.Forms
    Add-Type -AssemblyName UIAutomationClient
    Add-Type -AssemblyName UIAutomationTypes

    if (-not ("TFDStep1" -as [type])) {
        Add-Type @"
using System;
using System.Runtime.InteropServices;

public static class TFDStep1
{
    [DllImport("user32.dll")]
    public static extern bool SetForegroundWindow(IntPtr hWnd);
}
"@
    }

    $pidNow = (Get-Process TFDSafeCln | Select-Object -First 1).Id
    $desktop = [System.Windows.Automation.AutomationElement]::RootElement

    Write-Host ""
    Write-Host "========== STEP 1 ==========" -ForegroundColor Cyan
    Write-Host "PID = $pidNow"

    # 找 TMainForm
    $main = $desktop.FindFirst(
        [System.Windows.Automation.TreeScope]::Children,
        (New-Object System.Windows.Automation.AndCondition(
            (New-Object System.Windows.Automation.PropertyCondition(
                [System.Windows.Automation.AutomationElement]::ClassNameProperty,
                "TMainForm"
            )),
            (New-Object System.Windows.Automation.PropertyCondition(
                [System.Windows.Automation.AutomationElement]::ProcessIdProperty,
                $pidNow
            ))
        ))
    )

    if ($null -eq $main) {
        throw "找不到 TMainForm"
    }

    Write-Host "✓ 找到 TMainForm" -ForegroundColor Green

    $mainHwnd = [IntPtr]$main.Current.NativeWindowHandle

    [TFDStep1]::SetForegroundWindow($mainHwnd) | Out-Null
    Start-Sleep -Milliseconds 250

    # 已驗證路徑
    [System.Windows.Forms.SendKeys]::SendWait("{ESC}")
    Start-Sleep -Milliseconds 100

    [System.Windows.Forms.SendKeys]::SendWait("{F10}")
    Start-Sleep -Milliseconds 120

    [System.Windows.Forms.SendKeys]::SendWait("{DOWN}")
    Start-Sleep -Milliseconds 90

    [System.Windows.Forms.SendKeys]::SendWait("{DOWN}")
    Start-Sleep -Milliseconds 90

    [System.Windows.Forms.SendKeys]::SendWait("{ENTER}")
    Start-Sleep -Milliseconds 500

    # 驗證 TSFM030Form 是否真的出現
    $pidCond = New-Object System.Windows.Automation.PropertyCondition(
        [System.Windows.Automation.AutomationElement]::ProcessIdProperty,
        $pidNow
    )

    $all = $desktop.FindAll(
        [System.Windows.Automation.TreeScope]::Descendants,
        $pidCond
    )

    $placeForm = $null

    for ($i=0; $i -lt $all.Count; $i++) {

        $e = $all.Item($i)

        try {
            if ($e.Current.ClassName -eq "TSFM030Form") {
                $placeForm = $e
                break
            }
        }
        catch {}
    }

    if ($null -eq $placeForm) {
        throw "已送出快捷鍵，但找不到 TSFM030Form"
    }

    Write-Host ""
    Write-Host "★ STEP 1 成功" -ForegroundColor Green
    Write-Host "已進入場所建檔，找到 TSFM030Form。" -ForegroundColor Green
}

# ============================================================
# STEP 2：填入場所關鍵字
# ============================================================
$Steps['2'] = {
    Add-Type -AssemblyName UIAutomationClient
    Add-Type -AssemblyName UIAutomationTypes

    $ErrorActionPreference = "Stop"

    function Norm([string]$s) {
        if ($null -eq $s) {
            return ""
        }

        return (($s -replace "\s","").Trim())
    }

    try {
        Write-Host ""
        Write-Host "========== STEP 2 完整測試 ==========" -ForegroundColor Cyan

        $pidNow =
            (Get-Process TFDSafeCln | Select-Object -First 1).Id

        Write-Host "PID = $pidNow"

        $desktop =
            [System.Windows.Automation.AutomationElement]::RootElement

        $pidCond =
            New-Object System.Windows.Automation.PropertyCondition(
                [System.Windows.Automation.AutomationElement]::ProcessIdProperty,
                $pidNow
            )

        # ======================================================
        # 1. 找整個 TFDSafeCln UIA Tree
        # ======================================================

        $all =
            $desktop.FindAll(
                [System.Windows.Automation.TreeScope]::Descendants,
                $pidCond
            )

        # ======================================================
        # 2. 找 TSFM030Form
        # ======================================================

        $form = $null

        for ($i=0; $i -lt $all.Count; $i++) {

            $e =
                $all.Item($i)

            try {

                if (
                    $e.Current.ClassName -eq
                    "TSFM030Form"
                ) {

                    $form = $e
                    break
                }

            }
            catch {}
        }

        if ($null -eq $form) {
            throw "找不到 TSFM030Form"
        }

        Write-Host "1. ✓ 找到 TSFM030Form" -ForegroundColor Green

        # ======================================================
        # 3. 找「另一場所」TevCheckBox
        # ======================================================

        $controls =
            $form.FindAll(
                [System.Windows.Automation.TreeScope]::Descendants,
                [System.Windows.Automation.Condition]::TrueCondition
            )

        $another = $null

        for ($i=0; $i -lt $controls.Count; $i++) {

            $e =
                $controls.Item($i)

            try {

                if (
                    $e.Current.ClassName -eq "TevCheckBox" -and
                    (Norm $e.Current.Name) -eq
                    (Norm "另一場所")
                ) {

                    $another = $e
                    break
                }

            }
            catch {}
        }

        if ($null -eq $another) {
            throw "找不到『另一場所』TevCheckBox"
        }

        Write-Host "2. ✓ 找到『另一場所』TevCheckBox" -ForegroundColor Green

        # ======================================================
        # 4. 往前找 sibling TevTextEdit
        # ======================================================

        $walker =
            [System.Windows.Automation.TreeWalker]::RawViewWalker

        $cur =
            $another

        $textEdit =
            $null

        for ($i=0; $i -lt 30; $i++) {

            try {

                $cur =
                    $walker.GetPreviousSibling($cur)

            }
            catch {

                $cur = $null
            }

            if ($null -eq $cur) {
                break
            }

            try {

                if (
                    $cur.Current.ClassName -eq
                    "TevTextEdit"
                ) {

                    $textEdit = $cur
                    break
                }

            }
            catch {}
        }

        if ($null -eq $textEdit) {
            throw "找不到『另一場所』前方的 TevTextEdit"
        }

        Write-Host "3. ✓ 找到場所名稱 TevTextEdit" -ForegroundColor Green

        # ======================================================
        # 5. 找內層 TcxCustomInnerTextEdit
        # ======================================================

        $inner =
            $textEdit.FindFirst(
                [System.Windows.Automation.TreeScope]::Descendants,
                (New-Object System.Windows.Automation.PropertyCondition(
                    [System.Windows.Automation.AutomationElement]::ClassNameProperty,
                    "TcxCustomInnerTextEdit"
                ))
            )

        if ($null -eq $inner) {
            throw "TevTextEdit 內找不到 TcxCustomInnerTextEdit"
        }

        Write-Host "4. ✓ 找到 TcxCustomInnerTextEdit" -ForegroundColor Green

        # ======================================================
        # 6. 列出支援 Pattern
        # ======================================================

        Write-Host ""
        Write-Host "支援 Pattern："

        $patterns =
            $inner.GetSupportedPatterns()

        foreach ($p in $patterns) {

            Write-Host (
                "  - " +
                $p.ProgrammaticName
            )
        }

        # ======================================================
        # 7. 取得 ValuePattern
        # ======================================================

        $vp =
            $null

        try {

            $vp =
                $inner.GetCurrentPattern(
                    [System.Windows.Automation.ValuePattern]::Pattern
                )

        }
        catch {}

        if ($null -eq $vp) {
            throw "TcxCustomInnerTextEdit 不支援 ValuePattern"
        }

        Write-Host ""
        Write-Host "5. ✓ ValuePattern 可用" -ForegroundColor Green

        # ======================================================
        # 8. 讀目前內容
        # ======================================================

        $before =
            $vp.Current.Value

        Write-Host "原本內容：[$before]"

        # ======================================================
        # 9. SetValue 場所關鍵字
        # ======================================================

        $vp.SetValue($TFD.Keyword)

        Start-Sleep -Milliseconds 300

        Write-Host "6. ✓ 已執行 SetValue('$($TFD.Keyword)')" -ForegroundColor Green

        # ======================================================
        # 10. 重新從 UIA Tree 抓一次 inner
        # 避免舊 element cache
        # ======================================================

        $all2 =
            $desktop.FindAll(
                [System.Windows.Automation.TreeScope]::Descendants,
                $pidCond
            )

        $another2 = $null

        for ($i=0; $i -lt $all2.Count; $i++) {

            $e =
                $all2.Item($i)

            try {

                if (
                    $e.Current.ClassName -eq "TevCheckBox" -and
                    (Norm $e.Current.Name) -eq
                    (Norm "另一場所")
                ) {

                    $another2 = $e
                    break
                }

            }
            catch {}
        }

        if ($null -eq $another2) {
            throw "重新驗證時找不到『另一場所』"
        }

        $cur2 =
            $another2

        $textEdit2 =
            $null

        for ($i=0; $i -lt 30; $i++) {

            try {

                $cur2 =
                    $walker.GetPreviousSibling($cur2)

            }
            catch {

                $cur2 = $null
            }

            if ($null -eq $cur2) {
                break
            }

            try {

                if (
                    $cur2.Current.ClassName -eq
                    "TevTextEdit"
                ) {

                    $textEdit2 = $cur2
                    break
                }

            }
            catch {}
        }

        if ($null -eq $textEdit2) {
            throw "重新驗證時找不到 TevTextEdit"
        }

        $inner2 =
            $textEdit2.FindFirst(
                [System.Windows.Automation.TreeScope]::Descendants,
                (New-Object System.Windows.Automation.PropertyCondition(
                    [System.Windows.Automation.AutomationElement]::ClassNameProperty,
                    "TcxCustomInnerTextEdit"
                ))
            )

        if ($null -eq $inner2) {
            throw "重新驗證時找不到 TcxCustomInnerTextEdit"
        }

        $vp2 =
            $inner2.GetCurrentPattern(
                [System.Windows.Automation.ValuePattern]::Pattern
            )

        $after =
            $vp2.Current.Value

        Write-Host ""
        Write-Host "重新讀回：[$after]"

        # ======================================================
        # 11. 最終驗證
        # ======================================================

        if (
            (Norm $after) -ne
            (Norm $TFD.Keyword)
        ) {

            throw "SetValue 後讀回不是『$($TFD.Keyword)』"
        }

        Write-Host ""
        Write-Host "==========================================" -ForegroundColor Green
        Write-Host "★ STEP 2 成功" -ForegroundColor Green
        Write-Host "場所名稱欄已確定為：$($TFD.Keyword)" -ForegroundColor Green
        Write-Host "==========================================" -ForegroundColor Green
    }
    catch {
        Write-Host ""
        Write-Host "========== STEP 2 中止 ==========" -ForegroundColor Red

        Write-Host $_.Exception.Message -ForegroundColor Red

        Write-Host ""
        Write-Host $_.InvocationInfo.PositionMessage -ForegroundColor Yellow

        throw   # 一鍵版：讓整個流程停下來
    }
}

# ============================================================
# STEP 3：查詢資料
# ============================================================
$Steps['3'] = {
    Add-Type -AssemblyName UIAutomationClient
    Add-Type -AssemblyName UIAutomationTypes

    $ErrorActionPreference = "Stop"

    function Norm([string]$s) {
        if ($null -eq $s) { return "" }
        return (($s -replace "\s","").Trim())
    }

    try {
        Write-Host ""
        Write-Host "========== STEP 3 ==========" -ForegroundColor Cyan

        $pidNow =
            (Get-Process TFDSafeCln | Select-Object -First 1).Id

        $desktop =
            [System.Windows.Automation.AutomationElement]::RootElement

        $pidCond =
            New-Object System.Windows.Automation.PropertyCondition(
                [System.Windows.Automation.AutomationElement]::ProcessIdProperty,
                $pidNow
            )

        # ======================================================
        # 1. 找 TSFM030Form
        # ======================================================

        $all =
            $desktop.FindAll(
                [System.Windows.Automation.TreeScope]::Descendants,
                $pidCond
            )

        $form = $null

        for ($i=0; $i -lt $all.Count; $i++) {

            $e = $all.Item($i)

            try {
                if ($e.Current.ClassName -eq "TSFM030Form") {
                    $form = $e
                    break
                }
            }
            catch {}
        }

        if ($null -eq $form) {
            throw "找不到 TSFM030Form"
        }

        Write-Host "1. ✓ 找到 TSFM030Form" -ForegroundColor Green

        # ======================================================
        # 2. 找 Q.查詢資料 TevButton
        # ======================================================

        $buttons =
            $form.FindAll(
                [System.Windows.Automation.TreeScope]::Descendants,
                (New-Object System.Windows.Automation.PropertyCondition(
                    [System.Windows.Automation.AutomationElement]::ClassNameProperty,
                    "TevButton"
                ))
            )

        $query = $null

        for ($i=0; $i -lt $buttons.Count; $i++) {

            $b = $buttons.Item($i)

            try {

                if (
                    (Norm $b.Current.Name) -eq
                    (Norm "Q.查詢資料")
                ) {

                    $query = $b
                    break
                }
            }
            catch {}
        }

        if ($null -eq $query) {
            throw "找不到 Q.查詢資料 TevButton"
        }

        Write-Host "2. ✓ 找到 Q.查詢資料" -ForegroundColor Green

        # ======================================================
        # 3. Invoke
        # ======================================================

        $invoke =
            $query.GetCurrentPattern(
                [System.Windows.Automation.InvokePattern]::Pattern
            )

        $invoke.Invoke()

        Write-Host "3. ✓ 已 Invoke Q.查詢資料" -ForegroundColor Green

        # ======================================================
        # 4. 等待查詢結果 Tab + Grid
        # ======================================================

        $sw =
            [Diagnostics.Stopwatch]::StartNew()

        $resultTab = $null
        $grids = @()

        while ($sw.Elapsed.TotalSeconds -lt 15) {

            Start-Sleep -Milliseconds 250

            $all2 =
                $desktop.FindAll(
                    [System.Windows.Automation.TreeScope]::Descendants,
                    $pidCond
                )

            $resultTab = $null

            for ($i=0; $i -lt $all2.Count; $i++) {

                $e = $all2.Item($i)

                try {

                    if (
                        $e.Current.ClassName -eq "TevTabSheet" -and
                        (Norm $e.Current.Name) -eq
                        (Norm "查詢結果")
                    ) {

                        $resultTab = $e
                        break
                    }
                }
                catch {}
            }

            if ($null -ne $resultTab) {

                $grids =
                    @(
                        $resultTab.FindAll(
                            [System.Windows.Automation.TreeScope]::Descendants,
                            (New-Object System.Windows.Automation.PropertyCondition(
                                [System.Windows.Automation.AutomationElement]::ClassNameProperty,
                                "TcxGridSite"
                            ))
                        )
                    )

                if ($grids.Count -gt 0) {
                    break
                }
            }
        }

        if ($null -eq $resultTab) {
            throw "15 秒內找不到『查詢結果』TevTabSheet"
        }

        Write-Host "4. ✓ 找到『查詢結果』Tab" -ForegroundColor Green

        if ($grids.Count -eq 0) {
            throw "查詢結果內找不到 TcxGridSite"
        }

        Write-Host ""
        Write-Host "查詢結果內 Grid 數量 = $($grids.Count)"

        # ======================================================
        # 5. 列出 Grid 尺寸
        # ======================================================

        $largest = $null
        $bestArea = 0

        for ($i=0; $i -lt $grids.Count; $i++) {

            $g = $grids[$i]

            $r =
                $g.Current.BoundingRectangle

            $area =
                $r.Width * $r.Height

            Write-Host (
                "Grid[{0}] Rect={1},{2},{3},{4} Area={5}" -f
                $i,
                [int]$r.Left,
                [int]$r.Top,
                [int]$r.Width,
                [int]$r.Height,
                [int]$area
            )

            if ($area -gt $bestArea) {
                $bestArea = $area
                $largest = $g
            }
        }

        if ($null -eq $largest) {
            throw "無法判定最大 Grid"
        }

        $lr =
            $largest.Current.BoundingRectangle

        Write-Host ""
        Write-Host (
            "最大 Grid = {0},{1},{2},{3}" -f
            [int]$lr.Left,
            [int]$lr.Top,
            [int]$lr.Width,
            [int]$lr.Height
        ) -ForegroundColor Yellow

        Write-Host ""
        Write-Host "==========================================" -ForegroundColor Green
        Write-Host "★ STEP 3 成功" -ForegroundColor Green
        Write-Host "Q.查詢資料已執行，查詢結果與 Grid 已出現。" -ForegroundColor Green
        Write-Host "==========================================" -ForegroundColor Green
    }
    catch {
        Write-Host ""
        Write-Host "========== STEP 3 中止 ==========" -ForegroundColor Red
        Write-Host $_.Exception.Message -ForegroundColor Red
        Write-Host ""
        Write-Host $_.InvocationInfo.PositionMessage -ForegroundColor Yellow

        throw   # 一鍵版：讓整個流程停下來
    }
}

# ============================================================
# STEP 4：讀取查詢結果 Grid
# ============================================================

    Add-Type -AssemblyName UIAutomationClient
    Add-Type -AssemblyName UIAutomationTypes

    if (-not ("TFDStep4" -as [type])) {

using System;
using System.Runtime.InteropServices;

public static class TFDStep4
{
    [DllImport("user32.dll")]
    public static extern bool SetForegroundWindow(IntPtr hWnd);

    [DllImport("user32.dll")]

    [DllImport("user32.dll")]
    public static extern void mouse_event(
        uint flags,
        uint dx,
        uint dy,
        uint data,
        UIntPtr extraInfo
    );
}
"@
    }

    $ErrorActionPreference = "Stop"

    function Norm([string]$s) {
        if ($null -eq $s) { return "" }
        return (($s -replace "\s","").Trim())
    }

    try {
        Write-Host ""
        Write-Host "========== STEP 4 ==========" -ForegroundColor Cyan

        $pidNow =
            (Get-Process TFDSafeCln | Select-Object -First 1).Id

            [System.Windows.Automation.AutomationElement]::RootElement

        # ======================================================
        # 1. 找 TMainForm
        # ======================================================

                        "TMainForm"
                    )),
                    (New-Object System.Windows.Automation.PropertyCondition(

                        $pidNow
                    ))
                ))
            )

        if ($null -eq $main) {
            throw "找不到 TMainForm"
        }

        $mainHwnd =

        Write-Host "1. ✓ 找到 TMainForm" -ForegroundColor Green

        # ======================================================

        # ======================================================

        $resultTab =

        # ======================================================

        # ======================================================

        $grids =

                [int]$r.Height

                $bestArea = $area

        if ($null -eq $grid) {
            throw "無法判定最大 Grid"
        }

        $r =
            $grid.Current.BoundingRectangle

        Write-Host ""
        Write-Host (
            "3. ✓ 最大 Grid = {0},{1},{2},{3}" -f
            [int]$r.Left,
            [int]$r.Top,
            [int]$r.Width,
            [int]$r.Height
        ) -ForegroundColor Green

        # ======================================================
        # 4. 黃金 9 點探測
        # ======================================================

        $probePoints = @(
            @(0.17,0.12),
            @(0.25,0.12),
            @(0.35,0.12),

            @(0.17,0.16),
            @(0.25,0.16),
            @(0.35,0.16),

            @(0.17,0.20),
            @(0.25,0.20),
            @(0.35,0.20)
        )

        $goodRaw = ""

            (Get-Process TFDSafeCln | Select-Object -First 1).Id

        $desktop =
            [System.Windows.Automation.AutomationElement]::RootElement

        # ======================================================
        # 1. 找 TMainForm
        # ======================================================

        $main =
            $desktop.FindFirst(
                [System.Windows.Automation.TreeScope]::Children,
                (New-Object System.Windows.Automation.AndCondition(
                    (New-Object System.Windows.Automation.PropertyCondition(
                        [System.Windows.Automation.AutomationElement]::ClassNameProperty,
                        "TMainForm"
                    )),
                    (New-Object System.Windows.Automation.PropertyCondition(
                        [System.Windows.Automation.AutomationElement]::ProcessIdProperty,
                        $pidNow
                    ))
                ))
            )

        if ($null -eq $main) {
            throw "找不到 TMainForm"
        }

        $mainHwnd =
            [IntPtr]$main.Current.NativeWindowHandle

        # ======================================================
        # 2. 找查詢結果
        # ======================================================

        $resultTab =
            $main.FindFirst(
                [System.Windows.Automation.TreeScope]::Descendants,
                (New-Object System.Windows.Automation.AndCondition(
                    (New-Object System.Windows.Automation.PropertyCondition(
                        [System.Windows.Automation.AutomationElement]::ClassNameProperty,
                        "TevTabSheet"
                    )),
                    (New-Object System.Windows.Automation.PropertyCondition(
                        [System.Windows.Automation.AutomationElement]::NameProperty,
                        "查詢結果"
                    ))
                ))
            )

        if ($null -eq $resultTab) {
            throw "找不到查詢結果"
        }

        # ======================================================
        # 3. 最大 Grid
        # ======================================================

        $grids =
            $resultTab.FindAll(
                [System.Windows.Automation.TreeScope]::Descendants,
                (New-Object System.Windows.Automation.PropertyCondition(
                    [System.Windows.Automation.AutomationElement]::ClassNameProperty,
                    "TcxGridSite"
                ))
            )

        $grid = $null
        $bestArea = 0

        for ($i=0; $i -lt $grids.Count; $i++) {

            $g = $grids.Item($i)
            $r = $g.Current.BoundingRectangle
            $area = $r.Width * $r.Height

            if ($area -gt $bestArea) {
                $bestArea = $area
                $grid = $g
            }
        }

        if ($null -eq $grid) {
            throw "找不到最大 Grid"
        }

        # ======================================================
        # 4. 重新啟用 Grid
        # ======================================================

        [TFDStep7]::SetForegroundWindow(
            $mainHwnd
        ) | Out-Null

        Start-Sleep -Milliseconds 200

        $r =
            $grid.Current.BoundingRectangle

        $x =
            [int](
                $r.Left +
                ($r.Width * 0.17)
            )

        $y =
            [int](
                $r.Top +
                ($r.Height * 0.12)
            )

        [TFDStep7]::SetCursorPos(
            $x,
            $y
        ) | Out-Null

        Start-Sleep -Milliseconds 80

        [TFDStep7]::mouse_event(
            0x0002,0,0,0,[UIntPtr]::Zero
        )

        Start-Sleep -Milliseconds 60

        [TFDStep7]::mouse_event(
            0x0004,0,0,0,[UIntPtr]::Zero
        )

        Start-Sleep -Milliseconds 180

        $focused =
            [System.Windows.Automation.AutomationElement]::FocusedElement

        if ($focused.Current.ProcessId -ne $pidNow) {
            throw "重新啟用 Grid 後焦點不在安管"
        }

        # ======================================================
        # 5. 安全讀取 Grid
        # ======================================================

        function Read-GridRaw {

            $focused =
                [System.Windows.Automation.AutomationElement]::FocusedElement

            if ($focused.Current.ProcessId -ne $pidNow) {
                throw "Ctrl+C 前焦點已離開安管"
            }

            Set-Clipboard "__TFD_EMPTY__"

            [System.Windows.Forms.SendKeys]::SendWait("^c")

            Start-Sleep -Milliseconds 170

            $v = Get-Clipboard -Raw

            if (
                [string]::IsNullOrWhiteSpace($v) -or
                $v -eq "__TFD_EMPTY__"
            ) {
                return ""
            }

            return $v.Trim()
        }

        # ======================================================
        # 6. 回第一筆
        # ======================================================

        $previous = ""

        for ($i=0; $i -lt 300; $i++) {

            $current = Read-GridRaw

            if ([string]::IsNullOrWhiteSpace($current)) {
                throw "回第一筆途中讀不到 Grid"
            }

            if ($current -eq $previous) {
                break
            }

            $previous = $current

            [System.Windows.Forms.SendKeys]::SendWait("{UP}")

            Start-Sleep -Milliseconds 120
        }

        # ======================================================
        # 7. 走到指定序號
        # ======================================================

        for ($i=1; $i -lt $choice; $i++) {

            [System.Windows.Forms.SendKeys]::SendWait("{DOWN}")

            Start-Sleep -Milliseconds 120
        }

        # ======================================================
        # 8. 讀選定列
        # ======================================================

        $selected = Read-GridRaw

        if ([string]::IsNullOrWhiteSpace($selected)) {
            throw "選定列讀取失敗"
        }

        $lines = @(
            $selected -split "`r?`n" |
            Where-Object {
                -not [string]::IsNullOrWhiteSpace($_)
            }
        )

        if ($lines.Count -lt 2) {
            throw "選定列格式異常"
        }

        $headers = @($lines[0] -split "`t")
        $values = @($lines[1] -split "`t")

        $placeNo = ""
        $placeName = ""

        for ($c=0; $c -lt $headers.Count; $c++) {

            $h = Norm $headers[$c]
            $v = ""

            if ($c -lt $values.Count) {
                $v = $values[$c].Trim()
            }

            if ($h -eq "場所編號") {
                $placeNo = $v
            }

            if ($h -eq "場所名稱") {
                $placeName = $v
            }
        }

        Write-Host ""
        Write-Host "========================================" -ForegroundColor Cyan
        Write-Host "選定場所" -ForegroundColor Cyan
        Write-Host "場所編號：$placeNo"
        Write-Host "場所名稱：$placeName" -ForegroundColor Yellow
        Write-Host "========================================" -ForegroundColor Cyan
        Write-Host ""

        if (
            [string]::IsNullOrWhiteSpace($placeNo) -or
            [string]::IsNullOrWhiteSpace($placeName)
        ) {
            throw "無法解析選定列的場所編號／名稱"
        }

        # ======================================================
        # 9. 明確人工確認
        # ======================================================

        while ($true) {

            $confirm =
                Read-Host "確認這是正確場所？(Y/N)"

            $confirm =
                $confirm.Trim().ToUpperInvariant()

            if ($confirm -eq "Y") {
                break
            }

            if ($confirm -eq "N") {
                break
            }

            Write-Host "請輸入 Y 或 N" -ForegroundColor Yellow
        }

        if ($confirm -eq "N") {

            # 只退出本步，不進場所
            Write-Host ""
            Write-Host "已取消本次選擇，將重新執行 STEP 7。" -ForegroundColor Yellow

            $TFD.RetryChoice = $true
            return
        }

        $TFD.RetryChoice = $false
        $TFD.Choice = $choice
        $TFD.PlaceNo = $placeNo
        $TFD.PlaceName = $placeName

        # ======================================================
        # 10. ★ 只把 Grid 定位到正確列
        # ★ 不進場所
        # ======================================================

        Write-Host ""
        Write-Host "==========================================" -ForegroundColor Green
        Write-Host "★ STEP 7 成功" -ForegroundColor Green
        Write-Host "已確認場所：$placeNo  $placeName" -ForegroundColor Green
        Write-Host "目前仍停在查詢結果，尚未進入場所。" -ForegroundColor Green
        Write-Host "==========================================" -ForegroundColor Green
    }
    catch {

        Write-Host ""
        Write-Host "========== STEP 7 中止 ==========" -ForegroundColor Red
        Write-Host $_.Exception.Message -ForegroundColor Red
        Write-Host ""
        Write-Host $_.InvocationInfo.PositionMessage -ForegroundColor Yellow

        throw   # 一鍵版：讓整個流程停下來
    }
}

$Steps['8'] = {
    Add-Type -AssemblyName System.Windows.Forms
    Add-Type -AssemblyName UIAutomationClient
    Add-Type -AssemblyName UIAutomationTypes

    if (-not ("TFDStep8Final" -as [type])) {

        $main =
            $desktop.FindFirst(

                    (New-Object System.Windows.Automation.PropertyCondition(
                        [System.Windows.Automation.AutomationElement]::ClassNameProperty,

                    (New-Object System.Windows.Automation.PropertyCondition(
                        [System.Windows.Automation.AutomationElement]::ProcessIdProperty,
                        $pidNow

                ))

        if ($null -eq $main) {
            throw "找不到 TMainForm"
        }

        $mainHwnd =
            [IntPtr]$main.Current.NativeWindowHandle

        # 關鍵：Step 7 可能最小化；先還原
        [TFDStep8Final]::ShowWindow(

        Start-Sleep -Milliseconds 250

        [TFDStep8Final]::SetForegroundWindow(
            $mainHwnd

        Start-Sleep -Milliseconds 350

            $main.Current.BoundingRectangle

        Write-Host (
            "1. ✓ 安管已還原 Rect={0},{1},{2},{3}" -f

            [int]$mr.Width,
            [int]$mr.Height

                (New-Object System.Windows.Automation.AndCondition(
                    (New-Object System.Windows.Automation.PropertyCondition(
                        [System.Windows.Automation.AutomationElement]::ClassNameProperty,
                        "TevTabSheet"

                    (New-Object System.Windows.Automation.PropertyCondition(
                        [System.Windows.Automation.AutomationElement]::NameProperty,
                        "查詢結果"
                    ))

        if ($null -eq $resultTab) {
            throw "目前找不到『查詢結果』，Step 8 必須從 Step 7 完成後的查詢結果畫面執行"

        # 3. 最大 TcxGridSite
        # ======================================================

        $grids =
            $resultTab.FindAll(
                [System.Windows.Automation.TreeScope]::Descendants,

                    [System.Windows.Automation.AutomationElement]::ClassNameProperty,
                    "TcxGridSite"
                ))

        if ($grids.Count -eq 0) {
            throw "查詢結果內找不到 TcxGridSite"
        }

        $grid = $null
        $bestArea = 0

        for ($i=0; $i -lt $grids.Count; $i++) {

            $r = $g.Current.BoundingRectangle

            $area =
                $r.Width * $r.Height

                $grid = $g

        if ($null -eq $grid) {

            "2. ✓ 場所主 Grid={0},{1},{2},{3}" -f
            [int]$gr.Left,
            [int]$gr.Top,
            [int]$gr.Width,
            [int]$gr.Height
